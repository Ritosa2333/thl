cmake_minimum_required(VERSION 3.21)
project(thl_netlab LANGUAGES CXX)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT DEFINED LIBDATACHANNEL_SOURCE)
  message(FATAL_ERROR "Set LIBDATACHANNEL_SOURCE to the pinned dependency checkout")
endif()
add_subdirectory("${LIBDATACHANNEL_SOURCE}" datachannel EXCLUDE_FROM_ALL)
add_executable(thl-native-peer native_peer.cpp)
target_compile_features(thl-native-peer PRIVATE cxx_std_17)
target_include_directories(thl-native-peer PRIVATE "${LIBDATACHANNEL_SOURCE}/deps/json/single_include")
target_link_libraries(thl-native-peer PRIVATE datachannel)
